طريقة الحصول على رابط جدولك في Google Sheets (مرة واحدة فقط - دقيقتان)
1
افتح ملفك سجل إدارة مخزون ONBBS MACHINE SHOP في Google Sheets، ومن القائمة العلوية اختر: الإضافات (Extensions) ← Apps Script.
2
امسح أي كود موجود والصق الكود التالي، ثم اضغط حفظ:
// كود الربط السحابي المباشر لجدول: سجل إدارة مخزون ONBBS MACHINE SHOP // الصقه في: الإضافات (Extensions) > Apps Script ثم انشره كتطبيق ويب (Deploy > Web App) const SHEET_NAME = "مخزون الورشة"; function doGet(e) { const lock = LockService.getScriptLock(); lock.tryLock(5000); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0]; const lastRow = Math.max(sheet.getLastRow(), 10); const values = sheet.getRange(10, 2, lastRow - 9, 9).getValues(); // B10:J const items = []; for (let i = 0; i < values.length; i++) { const r = values[i]; if (r[0] && r[1]) { items.push({ code: String(r[0]).trim(), name: String(r[1]).trim(), category: String(r[2] || "غير مصنف").trim(), condition: String(r[3] || "صالح للاستخدام").trim(), qty: Number(r[4]) || 0, minQty: Number(r[5]) || 0, location: String(r[7] || "").trim(), notes: String(r[8] || "").trim() }); } } const meta = PropertiesService.getScriptProperties().getProperty("ONBBS_META"); const parsedMeta = meta ? JSON.parse(meta) : { updatedAt: Date.now(), updatedBy: "Google Sheets", logs: [] }; const payload = JSON.stringify({ ok: true, items: items, updatedAt: parsedMeta.updatedAt, updatedBy: parsedMeta.updatedBy, logs: parsedMeta.logs || [] }); if (e && e.parameter && e.parameter.callback) { return ContentService.createTextOutput(e.parameter.callback + "(" + payload + ");") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(payload).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const data = JSON.parse(e.postData.contents); const items = data.items || []; const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0]; // مسح الصفوف القديمة في جدول الأصناف B10:J const maxClear = Math.max(sheet.getLastRow(), 30); sheet.getRange(10, 2, maxClear - 9, 9).clearContent(); if (items.length > 0) { const rows = items.map(it => { const q = Number(it.qty) || 0; const m = Number(it.minQty) || 0; const st = q === 0 ? "نفذت الكمية" : (q <= m ? "منخفض - يلزم طلب" : "متوفر"); return [it.code, it.name, it.category, it.condition, q, m, st, it.location || "", it.notes || ""]; }); sheet.getRange(10, 2, rows.length, 9).setValues(rows); } // تحديث جدول ملخص التصنيفات الديناميكي في L10:N تلقائياً const catMap = {}; items.forEach(it => { const cats = String(it.category || "غير مصنف").split(/[,،]+/).map(s => s.trim()).filter(Boolean); cats.forEach(c => { if (!catMap[c]) catMap[c] = { count: 0, qty: 0 }; catMap[c].count += 1; catMap[c].qty += (Number(it.qty) || 0); }); }); sheet.getRange(10, 12, maxClear - 9, 3).clearContent(); const catRows = Object.keys(catMap).map(k => [k, catMap[k].count, catMap[k].qty]); if (catRows.length > 0) { sheet.getRange(10, 12, catRows.length, 3).setValues(catRows); } PropertiesService.getScriptProperties().setProperty("ONBBS_META", JSON.stringify({ updatedAt: data.updatedAt || Date.now(), updatedBy: data.updatedBy || "جوال", logs: (data.logs || []).slice(0, 25) })); return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
